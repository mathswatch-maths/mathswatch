(function () { const boardEl = document.getElementById('board'); const statusEl = document.getElementById('status'); const modeBtn = document.getElementById('mode-toggle'); const restartBtn = document.getElementById('restart'); let cells, current, gameOver, vsCpu; vsCpu = true; const lines = [ [0, 1, 2], [3, 4, 5], [6, 7, 8], [0, 3, 6], [1, 4, 7], [2, 5, 8], [0, 4, 8], [2, 4, 6], ]; function reset() { cells = Array(9).fill(null); current = 'X'; gameOver = false; render(); updateStatus(); } function render() { boardEl.innerHTML = ''; cells.forEach((val, i) => { const div = document.createElement('div'); div.className = 'ttt-cell'; div.textContent = val || ''; div.addEventListener('click', () => handleMove(i)); boardEl.appendChild(div); }); } function updateStatus(msg) { if (msg) { statusEl.textContent = msg; return; } statusEl.textContent = vsCpu ? (current === 'X' ?
